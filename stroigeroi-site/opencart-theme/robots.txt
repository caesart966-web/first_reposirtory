# robots.txt for the live site stroigeroi.ru. Goes to the site root,
# next to index.php - NOT into the theme folder.
#
# Built from the stock ocStore 3.0.3.7 robots.txt that was on the site
# on 23.09.2026 (all its rules are kept below) plus Clean-param for
# Yandex and the real sitemap address.
#
# ASCII only, on purpose: the server sends .txt without a charset,
# and browsers show any Russian text in this file as garbage.
# The reasons behind every rule are in stroigeroi-site/README.md
# ("Indeksaciya") and in opencart-theme/USTANOVKA.md, step 6.
#
# There is deliberately NO "Disallow: /index.php?route=" here.
# SEO URLs are OFF on this site: every page lives at index.php?route=...,
# and that popular rule would remove the whole shop from search.

User-agent: *

# Pages for one person: account, checkout, cart, comparison, search.
Disallow: /*route=account/
Disallow: /*route=affiliate/
Disallow: /*route=checkout/
Disallow: /*route=simplecheckout/
Disallow: /*route=common/
Disallow: /*route=product/compare
Disallow: /*route=product/search
Disallow: /*route=information/contact/success
Disallow: /*route=extension/module/live_search

# The same product opened through its manufacturer is a duplicate.
# The stock sitemap feed lists such links too; this keeps them out.
Disallow: /*route=product/product*&manufacturer_id=

# Engine internals.
Disallow: /admin/
Disallow: /system/
Disallow: /download/
Disallow: /catalog/controller/
Disallow: /catalog/language/
Disallow: /catalog/model/

# A catalogue page with another sort order or page size is the same
# page, not a new one. Same for search filters, tags and tracking.
Disallow: /*?sort=
Disallow: /*&sort=
Disallow: /*?order=
Disallow: /*&order=
Disallow: /*?limit=
Disallow: /*&limit=
Disallow: /*?filter=
Disallow: /*&filter=
Disallow: /*?filter_name=
Disallow: /*&filter_name=
Disallow: /*?filter_sub_category=
Disallow: /*&filter_sub_category=
Disallow: /*?filter_description=
Disallow: /*&filter_description=
Disallow: /*?tag=
Disallow: /*&tag=
Disallow: /*?tracking=
Disallow: /*&tracking=

# Catalogue pagination stays open: products on page 2 and further
# must stay reachable. A sorted page is still closed: its Disallow
# rule is longer, and the longer rule wins.
Allow: /*page=

# Styles, scripts and pictures. No "$" at the end on purpose:
# style.css is linked as style.css?v=4d04a862, and "$" would miss it.
Allow: /catalog/view/
Allow: /image/
Allow: /*.css
Allow: /*.js
Allow: /*.jpg
Allow: /*.jpeg
Allow: /*.png
Allow: /*.gif
Allow: /*.webp
Allow: /*.avif
Allow: /*.svg
Allow: /*.woff
Allow: /*.woff2

# Yandex: these parameters do not change the page, so the page stays
# in the index and its weight is not split between copies.
# "page" is NOT here on purpose: page 2 shows other products.
Clean-param: sort&order&limit&tracking /
Clean-param: utm_source&utm_medium&utm_campaign&utm_term&utm_content&yclid&gclid&_openstat /

# Sitemap: ocStore feed "Sitemap" (Extensions -> Promotion channels).
# Address checked in the admin panel on 23.09.2026.
Sitemap: https://stroigeroi.ru/index.php?route=extension/feed/google_sitemap
